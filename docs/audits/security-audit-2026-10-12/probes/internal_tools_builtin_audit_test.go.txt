package builtin
import (
 "context"
 "net/http"
 "net/http/httptest"
 "strings"
 "testing"
 "time"
)
func TestAuditHTTPSProxySendsPlaintextCredential(t *testing.T) {
 got := make(chan string, 1)
 proxy := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
  got <- r.Method + " " + r.Header.Get("Proxy-Authorization")
  w.WriteHeader(http.StatusBadGateway)
 }))
 defer proxy.Close()
 proxyURL := "https://dummy-user:dummy-password@" + strings.TrimPrefix(proxy.URL,"http://")
 tr := ssrfGuardedTransport(proxyURL)
 defer tr.CloseIdleConnections()
 c := &http.Client{Transport:tr, Timeout:3*time.Second}
 req, _ := http.NewRequestWithContext(context.Background(), "GET", "http://public.audit.invalid/", nil)
 if resp, _ := c.Do(req); resp != nil { resp.Body.Close() }
 select {
 case wire := <-got:
  if !strings.HasPrefix(wire,"CONNECT Basic ") { t.Fatalf("wire=%q",wire) }
  t.Log("CONFIRMED: https proxy received CONNECT and dummy Basic credentials as plaintext HTTP")
 case <-time.After(3*time.Second): t.Fatal("no plaintext CONNECT observed")
 }
}
