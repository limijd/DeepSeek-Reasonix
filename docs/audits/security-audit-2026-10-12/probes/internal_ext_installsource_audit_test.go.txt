package installsource
import (
 "io"
 "net/http"
 "net/http/httptest"
 "net/url"
 "net"
 "testing"
 "time"
)
func TestAuditProxyBypassesInstallSSRFGuard(t *testing.T) {
 target := make(chan string,1)
 proxy := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter,r *http.Request){
  target <- r.URL.Hostname()
  io.WriteString(w,"AUDIT_DUMMY_INTERNAL_RESPONSE")
 }))
 defer proxy.Close()
 u,_:=url.Parse(proxy.URL)
 tr:=&http.Transport{Proxy:http.ProxyURL(u)}
 defer tr.CloseIdleConnections()
 guarded:=ssrfGuardClient(&http.Client{Transport:tr,Timeout:3*time.Second})
 if !blockedFetchIP(net.ParseIP("169.254.169.254")) { t.Fatal("target is not on blocklist") }
 res,err:=guarded.Get("http://169.254.169.254/audit-only")
 if err!=nil { t.Fatal(err) }
 defer res.Body.Close()
 b,_:=io.ReadAll(res.Body)
 if string(b)!="AUDIT_DUMMY_INTERNAL_RESPONSE" { t.Fatalf("body=%q",b) }
 if got:=<-target;got!="169.254.169.254" { t.Fatalf("target=%q",got) }
 t.Log("CONFIRMED: blocked metadata IP was forwarded through a local fake proxy; no actual metadata service contacted")
}
